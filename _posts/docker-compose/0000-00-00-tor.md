---

*Run a Tor SOCKS proxy with custom exit-node exclusions.*

```yaml
services:
  tor:
    restart: 'always'
    container_name: 'tor'
    build: '.'
    ports:
      - '127.0.0.1:9050:9050'
```

---

*Dockerfile:*

```dockerfile
FROM alpine:latest

RUN apk add --no-cache tor && \
    rm -rf /var/cache/apk/* /tmp/* /var/tmp/*

ADD torrc /etc/tor/torrc

CMD ["/usr/bin/tor"]
```

---

*torrc:*

```text
SocksPort 0.0.0.0:9050
ExcludeExitNodes {ru}, {ua}, {by}
```
